### Create API Routes for Lead Management

**File: `backend/routes/leads.js`**

```javascript
const express = require('express');
const router = express.Router();
const { google } = require('googleapis');
const twilio = require('twilio');
const axios = require('axios');

const sheets = google.sheets('v4');
const SPREADSHEET_ID = 'YOUR_GOOGLE_SHEET_ID'; // Create a Google Sheet for leads
const VAPI_API_KEY = 'YOUR_VAPI_API_KEY_HERE';
const TWILIO_ACCOUNT_SID = 'YOUR_TWILIO_SID';
const TWILIO_AUTH_TOKEN = 'YOUR_TWILIO_TOKEN';
const TWILIO_PHONE = '+1234567890'; // Your Twilio number

// Initialize services
const twilioClient = twilio(TWILIO_ACCOUNT_SID, TWILIO_AUTH_TOKEN);

// Save lead to Google Sheets
router.post('/save-lead', async (req, res) => {
  try {
    const { name, phone, email, loanType, amount, city, callId, timestamp } = req.body;

    const values = [
      [
        name,
        phone,
        email,
        loanType,
        amount,
        city || 'Latur',
        new Date(timestamp || Date.now()).toLocaleString('en-IN'),
        callId || 'N/A',
        'Active'
      ]
    ];

    const request = {
      spreadsheetId: SPREADSHEET_ID,
      range: 'Leads!A1',
      valueInputOption: 'USER_ENTERED',
      resource: { values },
      auth: new google.auth.GoogleAuth({
        keyFile: 'path/to/service-account-key.json',
        scopes: ['https://www.googleapis.com/auth/spreadsheets']
      })
    };

    await sheets.spreadsheets.values.append(request);

    res.json({ success: true, message: 'Lead saved successfully' });
  } catch (error) {
    console.error('Error saving lead:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

// Send WhatsApp message
router.post('/send-whatsapp', async (req, res) => {
  try {
    const { phoneNumber, message, loanType } = req.body;

    // Using Twilio WhatsApp API or your WhatsApp Business API
    const result = await twilioClient.messages.create({
      from: 'whatsapp:+1234567890', // Twilio WhatsApp number
      to: `whatsapp:${phoneNumber}`,
      body: message
    });

    res.json({ success: true, messageSid: result.sid });
  } catch (error) {
    console.error('Error sending WhatsApp:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

// Send SMS reminder
router.post('/send-sms', async (req, res) => {
  try {
    const { phoneNumber, message } = req.body;

    const result = await twilioClient.messages.create({
      from: '+1234567890', // Your Twilio number
      to: phoneNumber,
      body: message
    });

    res.json({ success: true, messageSid: result.sid });
  } catch (error) {
    console.error('Error sending SMS:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

// Handle VAPI webhook callbacks
router.post('/webhooks/vapi-callback', async (req, res) => {
  try {
    const { callId, callStatus, transcript, analysis, phoneNumber } = req.body;

    // Extract call data
    const callData = {
      callId,
      phoneNumber,
      callStatus,
      transcript: transcript || 'N/A',
      extractedData: analysis?.structuredData || {},
      timestamp: new Date().toISOString()
    };

    // Save to Google Sheets
    await saveLeadCallData(callData);

    // Send follow-up WhatsApp if qualified
    if (analysis?.structuredData?.qualification_status === 'qualified') {
      await sendQualifiedLeadFollowup(phoneNumber, callData);
    }

    res.json({ success: true, message: 'Webhook processed' });
  } catch (error) {
    console.error('Webhook error:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

// Schedule appointment
router.post('/schedule-appointment', async (req, res) => {
  try {
    const { leadName, phoneNumber, loanType, appointmentDate, appointmentTime } = req.body;

    // Send SMS reminder
    await twilioClient.messages.create({
      from: '+1234567890',
      to: phoneNumber,
      body: `Hi ${leadName}! Your appointment for ${loanType} is scheduled on ${appointmentDate} at ${appointmentTime}. Call: +91 7249108474`
    });

    res.json({ success: true, message: 'Appointment scheduled and reminder sent' });
  } catch (error) {
    console.error('Appointment scheduling error:', error);
    res.status(500).json({ success: false, error: error.message });
  }
});

module.exports = router;
```

| Factor | Points | Criteria |
|--------|--------|----------|
| Monthly Income | 20 | ₹15,000+ = 20 pts |
| Employment Type | 15 | Salaried = 15, Self-employed = 10 |
| Age | 15 | 21-40 = 15, 41-50 = 10, 51-60 = 5 |
| Existing EMI | 15 | <30% of income = 15, 30-50% = 10, >50% = 0 |
| Credit History | 20 | Good = 20, Fair = 10, Poor = 0 |
| Loan Amount | 15 | Reasonable = 15, Excessive = 0 |

**Scoring Logic:**
- 85-100: Highly Qualified → Auto-approve
- 70-85: Qualified → Manual review
- 50-70: Needs Review → Request more docs
- <50: Not Qualified → Offer alternative
